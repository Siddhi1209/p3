
function status_queue() {

    // Beep flag
    $beepRequired = false;

    // (1) Read the data file safely (no shell_exec)
    $filePath = $this->currpath . "status_queue_8app.txt." . $this->datecrr;
    $rawLines = @file($filePath, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES);

    if (!$rawLines || !is_array($rawLines) || count($rawLines) === 0) {
        echo "<tr><td colspan='16' class='text-center'>No data (file missing or empty)</td></tr>";
        return;
    }

    // (2) Clean + prepare lines
    // Remove BOM, trim, replace NQKE with -1, hide VV3Q
    $data1 = [];

    foreach ($rawLines as $line) {

        // Remove BOM and extra spaces
        $line = preg_replace('/^\xEF\xBB\xBF/', '', $line);
        $line = trim($line);

        if ($line === '') {
            continue;
        }

        // Get queue name (first field)
        $parts = explode(',', $line, 2);
        $qname = trim($parts[0]);

        // Hide VV3Q (case-insensitive)
        if (strcasecmp($qname, 'VV3Q') === 0) {
            continue;
        }

        // Replace NQKE with -1
        $data1[] = str_replace('NQKE', '-1', $line);
    }

    if (empty($data1)) {
        echo "<tr><td colspan='16' class='text-center'>No valid queue data</td></tr>";
        return;
    }

    // (3) Transform rows (apply throwRed markup)
    $data = $this->throwRed($data1, ',');

    // Fallback if throwRed filters out too much
    if (empty($data)) {
        $data = $data1;
    }

    // (4) Render table
    foreach ($data as $row) {

        if ($row === '') {
            continue;
        }

        $cells = explode(',', $row);

        echo "<tr>";

        foreach ($cells as $k2 => $v2) {

            $v2t = trim((string)$v2);
            $plain = trim(html_entity_decode(strip_tags($v2t)));

            // Convert numeric value
            $v2n = preg_match('/^-?\d+$/', $plain) ? (int)$plain : null;

            // Queue name
            if ($k2 === 0) {

                // Queue name with popup link
                $link = "window.open('queuereplica.php?queue=" .
                        urlencode($plain) .
                        "&&date=" .
                        $this->datecrr .
                        "','','height=800,width=1400,scrollbars=yes,resizable=yes');";

                echo "<th><a onclick=\"$link\">" .
                     htmlspecialchars($plain, ENT_QUOTES) .
                     "</a></th>";

                continue;
            }

            // NQKE or -1 = thumbs down
            if ($plain === 'NQKE' || $v2n === -1) {

                echo "<td><i class='fa fa-thumbs-down' style='color:red'></i></td>";

                continue;
            }

            // 1 to 98 = display 0
            if ($v2n !== null && $v2n >= 1 && $v2n < 99) {

                echo "<td title='" .
                     htmlspecialchars($plain, ENT_QUOTES) .
                     "'>0</td>";

                continue;
            }

            // 99 to 499 = yellow
            if ($v2n !== null && $v2n >= 99 && $v2n < 500) {

                echo "<td class='bg-yellow text-center'>" .
                     htmlspecialchars($plain, ENT_QUOTES) .
                     "</td>";

                continue;
            }

            // 500 or greater = RED + BEEP
            if ($v2n !== null && $v2n >= 500) {

                // Set beep flag
                $beepRequired = true;

                echo "<td class='bg-red-active'>" .
                     htmlspecialchars($plain, ENT_QUOTES) .
                     "</td>";

                continue;
            }

            // Default
            echo "<td>" .
                 htmlspecialchars($plain, ENT_QUOTES) .
                 "</td>";
        }

        echo "</tr>";
    }

    // (5) Play beep if any buildup value is 500 or greater
    if ($beepRequired) {

        echo "
        <audio id='queueBeep' preload='auto'>
            <source src='/pace/Online_portal/includes/beep.mp3' type='audio/mpeg'>
        </audio>

        <script>
            (function() {

                var beep = document.getElementById('queueBeep');

                if (beep) {

                    beep.volume = 1.0;

                    var playBeep = function() {

                        beep.play().catch(function(error) {

                            console.log(
                                'Beep autoplay blocked by browser:',
                                error
                            );

                        });

                    };

                    // Try to play immediately
                    playBeep();

                    // Try again when user clicks on page
                    document.addEventListener(
                        'click',
                        playBeep,
                        { once: true }
                    );
                }

            })();
        </script>
        ";
    }
}
