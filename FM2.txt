public static function NT()
{
        $mithi_mountpoint = file("/opt/hpws/apache/cgi-bin/trials/bip/data/TESTING/NTlog.txt");

        foreach ($mithi_mountpoint as $line_num => $line1) {

                $varmount = explode(",", $line1);

                echo "<tr>";

                foreach ($varmount as $k3 => $v3) {

                        $v3 = trim($v3);

                        // TIME column
                        if ($k3 == 1) {

                                $logTime = strtotime($v3);
                                $currentTime = time();

                                // Difference in seconds
                                $diff = $currentTime - $logTime;

                                // 15 minutes or more = RED
                                if ($diff >= 900) {

                                        echo "<td class='text-center' style='background-color:red;color:white;font-weight:bold;'>"
                                             . $v3 .
                                             "</td>";

                                } else {

                                        echo "<td class='text-center'>"
                                             . $v3 .
                                             "</td>";
                                }

                        } else {

                                // LOG NAME column
                                echo "<td class='text-center'>"
                                     . $v3 .
                                     "</td>";
                        }
                }

                echo "</tr>";
        }
}
