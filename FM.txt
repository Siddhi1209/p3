/*
---------------------------------------------------------
READ COMMA-SEPARATED FILE
---------------------------------------------------------
*/

function readCSVFile($file)
{
    $rows = array();

    /*
    Check whether today's file exists
    */

    if (!file_exists($file)) {
        return $rows;
    }


    /*
    Open file
    */

    $handle = fopen($file, "r");


    if ($handle === false) {
        return $rows;
    }


    /*
    Read every CSV line
    */

    while (($data = fgetcsv($handle, 0, ",")) !== false) {

        /*
        Ignore completely blank lines
        */

        $hasData = false;


        foreach ($data as $value) {

            if (trim($value) !== "") {

                $hasData = true;

                break;
            }
        }


        if ($hasData) {

            $rows[] = $data;
        }
    }


    /*
    Close file
    */

    fclose($handle);


    return $rows;
}

/*
=========================================================
SPACE UTILIZATION CELL
=========================================================

If value > 75:
    RED BACKGROUND
    WHITE TEXT

If value <= 75:
    NORMAL

If value is "-":
    DISPLAY "-"

=========================================================
*/

function spaceCell($value)
{
    $value = trim($value);


    /*
    Empty or unavailable value
    */

    if ($value === "" || $value === "-") {

        return '<td class="space-cell dash">-</td>';
    }


    /*
    Remove % sign if present
    */

    $numericValue = intval(str_replace("%", "", $value));


    /*
    Greater than 75 = RED
    */

    if ($numericValue > 75) {

        return '<td class="space-cell high">'
             . h($numericValue)
             . '</td>';
    }


    /*
    75 or below = NORMAL
    */

    return '<td class="space-cell">'
         . h($numericValue)
         . '</td>';
}
