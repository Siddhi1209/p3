/*
---------------------------------------------------------
READ CSV FILE
---------------------------------------------------------
*/

function readCSVFile($file)
{
    $rows = array();

    if (!file_exists($file)) {
        return $rows;
    }

    $handle = fopen($file, "r");

    if ($handle === false) {
        return $rows;
    }

    while (($data = fgetcsv($handle, 0, ",")) !== false) {

        /*
        Ignore completely blank lines
        */

        $hasData = false;

        foreach ($data as $value) {

            if (trim($value) !== "") {

                $hasData = true;
                break;
            }
        }

        if ($hasData) {

            $rows[] = $data;
        }
    }

    fclose($handle);

    return $rows;
}

/*
---------------------------------------------------------
SPACE UTILIZATION CELL

If value > 75:
    RED BACKGROUND + WHITE TEXT

75 or below:
    NORMAL

"-":
    NORMAL DASH
---------------------------------------------------------
*/

function spaceCell($value)
{
    $value = trim($value);

    /*
    Empty / unavailable value
    */

    if ($value === "" || $value === "-") {

        return '<td class="space-cell dash">-</td>';
    }


    /*
    Remove % sign if present
    */

    $numericValue = intval(str_replace("%", "", $value));


    /*
    More than 75 = RED
    */

    if ($numericValue > 75) {

        return '<td class="space-cell high">'
             . h($numericValue)
             . '</td>';
    }


    /*
    75 or below = NORMAL
    */

    return '<td class="space-cell">'
         . h($numericValue)
         . '</td>';
}
