<!-- QUEUE BUILDUP BEEP START -->

<audio id="queueBuildupBeep"
       src="beep.mp3"
       preload="auto"></audio>

<script type="text/javascript">

var queueBeepInterval = null;
var queueBeepUnlocked = false;


/* ==========================================
   UNLOCK AUDIO AFTER USER INTERACTION
   ========================================== */

function unlockQueueBeep() {

    if (queueBeepUnlocked) {
        return;
    }

    var beep = document.getElementById("queueBuildupBeep");

    if (!beep) {
        console.log("QUEUE BEEP: beepSound not found");
        return;
    }

    beep.currentTime = 0;

    beep.play().then(function() {

        beep.pause();
        beep.currentTime = 0;

        queueBeepUnlocked = true;

        console.log("QUEUE BEEP: AUDIO ENABLED");

        checkQueueBeep();

    }).catch(function(error) {

        console.log("QUEUE BEEP: AUDIO BLOCKED");
        console.log(error);

    });
}


/* ==========================================
   CHECK WHETHER 500+ VALUE EXISTS
   ========================================== */

function checkQueueBeep() {

    var trigger =
        document.getElementById("queueBeepTrigger");

    if (trigger) {

        console.log("QUEUE BEEP: 500+ FOUND");

        if (queueBeepUnlocked) {

            startQueueBeep();

        }

    } else {

        stopQueueBeep();
    }
}


/* ==========================================
   START BEEP
   ========================================== */

function startQueueBeep() {

    if (queueBeepInterval !== null) {
        return;
    }

    var beep =
        document.getElementById("queueBuildupBeep");

    if (!beep) {
        console.log("QUEUE BEEP: audio not found");
        return;
    }

    console.log("QUEUE BEEP: STARTED");


    /* First beep immediately */

    beep.pause();
    beep.currentTime = 0;

    beep.play().catch(function(error) {

        console.log("QUEUE BEEP PLAY ERROR");
        console.log(error);

    });


    /* Repeat every 10 seconds */

    queueBeepInterval = setInterval(function() {

        var trigger =
            document.getElementById("queueBeepTrigger");

        /* Stop if 500+ no longer exists */

        if (!trigger) {

            stopQueueBeep();

            return;
        }


        beep.pause();
        beep.currentTime = 0;

        beep.play().catch(function(error) {

            console.log("QUEUE BEEP PLAY ERROR");
            console.log(error);

        });

        console.log("QUEUE BEEP: PLAY");

    }, 10000);
}


/* ==========================================
   STOP BEEP
   ========================================== */

function stopQueueBeep() {

    if (queueBeepInterval !== null) {

        clearInterval(queueBeepInterval);

        queueBeepInterval = null;
    }

    var beep =
        document.getElementById("queueBuildupBeep");

    if (beep) {

        beep.pause();
        beep.currentTime = 0;
    }

    console.log("QUEUE BEEP: STOPPED");
}


/* ==========================================
   FIRST USER INTERACTION
   ========================================== */

document.addEventListener(
    "click",
    unlockQueueBeep,
    { once: true }
);

document.addEventListener(
    "keydown",
    unlockQueueBeep,
    { once: true }
);


/* ==========================================
   INITIAL CHECK
   ========================================== */

document.addEventListener(
    "DOMContentLoaded",
    function() {
        checkQueueBeep();
    }
);

</script>

<!-- QUEUE BUILDUP BEEP END -->
