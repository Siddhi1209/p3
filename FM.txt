<?php

$count = 0;

if(file_exists($file))
{

    $fp = fopen($file,"r");

    $firstLine = true;

    while(($line = fgets($fp)) !== false)
    {

        $line = trim($line);

        if($line == "")
            continue;

        if($firstLine)
        {
            $firstLine = false;
            continue;
        }

        $data = explode(",", $line);

        echo "<tr>";

        $logType = trim($data[0]);

        if($logType=="NOPUSH")
        {
            echo "<td class='nopush'>$logType</td>";
        }
        elseif($logType=="FAILED")
        {
            echo "<td class='failed'>$logType</td>";
        }
        elseif($logType=="IGNORED")
        {
            echo "<td class='ignored'>$logType</td>";
        }
        else
        {
            echo "<td>$logType</td>";
        }

        for($i=1;$i<7;$i++)
        {
            echo "<td>";

            if(isset($data[$i]))
                echo htmlspecialchars($data[$i]);
            else
                echo "&nbsp;";

            echo "</td>";
        }

        echo "</tr>";

        $count++;

    }

    fclose($fp);
    }

}
else
{

    echo "<tr>";
    echo "<td colspan='7' class='message'>";
    echo "Report File Not Available.";
    echo "</td>";
    echo "</tr>";

}
