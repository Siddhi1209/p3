
document.getElementById("enableQueueBeep").onclick = function () {

    try {

        window.queueAudioContext =
            new (window.AudioContext || window.webkitAudioContext)();

        window.queueAudioContext.resume().then(function () {

            window.queueBeepEnabled = true;

            document.getElementById("enableQueueBeep").innerHTML =
                "QUEUE ALERT ENABLED";

            document.getElementById("enableQueueBeep").style.background =
                "green";

            console.log("QUEUE ALERT ENABLED");

        });

    } catch (error) {

        console.log("QUEUE AUDIO ERROR");
        console.log(error);

    }

};

<script type="text/javascript">

window.queueBeepEnabled = false;

document.getElementById("enableQueueBeep").onclick = function () {

    var beepAudio =
        document.getElementById("queueBuildupBeep");

    if (!beepAudio) {
        alert("Beep audio not found");
        return;
    }

    beepAudio.currentTime = 0;

    beepAudio.play().then(function () {

        beepAudio.pause();
        beepAudio.currentTime = 0;

        window.queueBeepEnabled = true;

        document.getElementById("enableQueueBeep").innerHTML =
            "QUEUE ALERT ENABLED";

        document.getElementById("enableQueueBeep").style.background =
            "green";

        console.log("QUEUE ALERT ENABLED");

    }).catch(function (error) {

        console.log("QUEUE ALERT ENABLE ERROR");
        console.log(error);

    });

};

</script>

<!-- END QUEUE ALERT ENABLE BUTTON -->
