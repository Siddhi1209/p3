<script type="text/javascript">

(function () {

    var beepAudio = document.getElementById("queueBuildupBeep");

    if (!beepAudio) {
        console.log("QUEUE BEEP: Audio element not found");
        return;
    }

    beepAudio.volume = 1.0;

    var audioUnlocked = false;

    // Unlock audio after user clicks on the portal
    function unlockQueueBeep() {

        if (audioUnlocked) {
            return;
        }

        beepAudio.play().then(function () {

            beepAudio.pause();
            beepAudio.currentTime = 0;
            audioUnlocked = true;

            console.log("QUEUE BEEP: Audio unlocked");

        }).catch(function () {

            console.log("QUEUE BEEP: Waiting for user click");

        });
    }

    document.addEventListener("click", unlockQueueBeep, false);
    document.addEventListener("keydown", unlockQueueBeep, false);


    // Function to play beep
    function playQueueBuildupBeep() {

        beepAudio.pause();
        beepAudio.currentTime = 0;

        beepAudio.play().then(function () {

            console.log("QUEUE BEEP: Playing");

        }).catch(function (error) {

            console.log("QUEUE BEEP: Playback blocked");
            console.log(error);

        });
    }


    // Check for trigger generated by function_apr.php
    function checkQueueBeep() {

        var trigger = document.querySelector(
            "#alldiv .queueBeepTrigger"
        );

        if (trigger) {

            console.log("QUEUE BEEP: 500 OR GREATER DETECTED");

            playQueueBuildupBeep();

            trigger.remove();
        }
    }


    // Monitor #alldiv because your portal replaces
    // its contents during automatic refresh
    var alldiv = document.getElementById("alldiv");

    if (alldiv && typeof MutationObserver !== "undefined") {

        var observer = new MutationObserver(function () {

            checkQueueBeep();

        });

        observer.observe(alldiv, {
            childList: true,
            subtree: true
        });

    }


    // Backup check every second
    setInterval(function () {

        checkQueueBeep();

    }, 1000);


})();

</script>
