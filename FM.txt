
<audio id="queueBuildupBeep" preload="auto">
    <source src="beep.mp3" type="audio/mpeg">
</audio>

<script>

var queueBeepUnlocked = true;

function checkQueueBeep() {

    if (!queueBeepUnlocked) {
        return;
    }

    var beep = document.getElementById("queueBuildupBeep");

    if (!beep) {
        console.log("QUEUE BEEP: AUDIO NOT FOUND");
        return;
    }

    var criticalQueue = document.querySelector(".beep-on-load");

    if (!criticalQueue) {
        return;
    }

    beep.currentTime = 0;

    beep.play().then(function () {

        console.log("QUEUE BEEP: PLAYING");

    }).catch(function (error) {

        console.log("QUEUE BEEP ERROR:");
        console.log(error);

    });
}


/* Check automatically every 5 seconds */
setInterval(function () {

    checkQueueBeep();

}, 5000);


/* Check immediately when page loads */
checkQueueBeep();

</script>
